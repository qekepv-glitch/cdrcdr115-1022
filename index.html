# 1. 資安最高指導原則 (絕不 Hardcode)
* 絕對不要在程式碼中寫死 Google Sheet ID 與 Google Drive Folder ID。
* 請一律使用 PropertiesService.getScriptProperties() 讀取 SHEET_ID 與 FOLDER_ID。並且在『code.gs』使用寫入的 property 也使用 SHEET_ID 與 FOLDER_ID 一致名稱，避免造成讀取不到。

# 2. 系統功能與介面
* 表單欄位：用餐日期、選擇班級、新增班級、來賓代表姓名、用餐人數、收費金額(自動計算：人數*50元)、經手人。
* 收據繪製：即時渲染明細 (代表姓名、人數、金額、經手人、日期)。
* 按鈕功能：下載 PNG 收據、下載 EXCEL (表單資料)、同步至表單與硬碟 (上傳收據圖檔至雲端並寫入表單)。

# 3. 全自動初始化機制 (getInitialData)
* 在網頁載入讀取資料 getInitialData 時加入自動偵測：若發現沒有對應的「午餐紀錄」與「班級列表」分頁，請自動建立分頁、寫入標題欄位，並填入模擬資料。不需要手動進去編輯器按「執行」。

const properties = PropertiesService.getScriptProperties();
const SHEET_ID = properties.getProperty("SHEET_ID");
const FOLDER_ID = properties.getProperty("FOLDER_ID");

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('來賓午餐登記與收據系統')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

// 網頁載入時觸發：自動建置資料表與回傳初始資料
function getInitialData() {
  if (!SHEET_ID || !FOLDER_ID) {
    return { error: "系統錯誤：請先至 GAS 後台設定 SHEET_ID 與 FOLDER_ID" };
  }

  const ss = SpreadsheetApp.openById(SHEET_ID);
  
  // 1. 自動建置班級列表
  let classSheet = ss.getSheetByName("班級列表");
  if (!classSheet) {
    classSheet = ss.insertSheet("班級列表");
    classSheet.appendRow(["班級名稱"]);
    classSheet.appendRow(["1638班"]); // 模擬資料
    classSheet.appendRow(["1663班"]); // 模擬資料
  }

  // 2. 自動建置午餐紀錄表
  let recordSheet = ss.getSheetByName("午餐紀錄");
  if (!recordSheet) {
    recordSheet = ss.insertSheet("午餐紀錄");
    recordSheet.appendRow(["時間印記", "用餐日期", "班級", "來賓代表", "人數", "金額", "經手人", "雲端收據連結"]);
  }

  // 取得班級選單資料
  const lastRow = classSheet.getLastRow();
  let classes = [];
  if (lastRow > 1) {
    classes = classSheet.getRange(2, 1, lastRow - 1, 1).getValues().flat().filter(String);
  }

  return { 
    classes: classes, 
    sheetId: SHEET_ID 
  };
}

// 接收前端資料：同步至表單與硬碟
function saveData(data) {
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const recordSheet = ss.getSheetByName("午餐紀錄");
    
    // 如果有新增班級，自動寫入班級列表
    if (data.isNewClass) {
      const classSheet = ss.getSheetByName("班級列表");
      classSheet.appendRow([data.className]);
    }

    let fileUrl = "";
    // 將 Base64 圖片轉檔並存入 Google Drive
    if (data.receiptImage) {
      const folder = DriveApp.getFolderById(FOLDER_ID);
      const base64Data = data.receiptImage.split(",")[1];
      const blob = Utilities.newBlob(Utilities.base64Decode(base64Data), "image/png", `${data.date}_${data.repName}_午餐收據.png`);
      const file = folder.createFile(blob);
      fileUrl = file.getUrl();
    }

    // 寫入試算表
    recordSheet.appendRow([
      new Date(), data.date, data.className, data.repName, data.pax, data.amount, data.handler, fileUrl
    ]);

    return { success: true };
  } catch (error) {
    return { success: false, message: error.toString() };
  }
}

// 取得系統設定 (決定是否顯示 Excel 按鈕)
function getSystemSettings() {
  try {
    const props = PropertiesService.getScriptProperties();
    const sheetId = props.getProperty('SHEET_ID');
    
    if (!sheetId) {
      return { success: false, error: "尚未在指令碼屬性中設定 SHEET_ID" };
    }

    const ss = SpreadsheetApp.openById(sheetId);
    let settingsSheet = ss.getSheetByName('系統設定');

    // 若尚未建立系統設定表，程式自動幫忙建立並填入預設值
    if (!settingsSheet) {
      settingsSheet = ss.insertSheet('系統設定');
      settingsSheet.getRange('A1:B2').setValues([
        ['設定項目', '設定值'],
        ['顯示下載EXCEL按鈕', true]
      ]);
      // 標題加粗加底色
      settingsSheet.getRange('A1:B1').setFontWeight('bold').setBackground('#f3f3f3');
    }

    // 讀取 B2 欄位的值
    const val = settingsSheet.getRange('B2').getValue();
    // 判斷是否為 true, TRUE, 或 "是"
    const isTrue = (val === true || val === 'TRUE' || val === 'true' || val === '是');
    
    return { success: true, showExcel: isTrue };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

function submitData(data) {
  try {
    const props = PropertiesService.getScriptProperties();
    const sheetId = props.getProperty('SHEET_ID');
    const folderId = props.getProperty('FOLDER_ID');

    if (!sheetId || !folderId) {
      throw new Error("系統安全參數未設定完整，請檢查 SHEET_ID 或 FOLDER_ID");
    }

    const ss = SpreadsheetApp.openById(sheetId);
    const rootFolder = DriveApp.getFolderById(folderId);

    // 1. 處理資料夾歸檔 (以學生座號建立/尋找子資料夾)
    const seatStr = "座號_" + data.seat;
    let subFolder;
    const folders = rootFolder.getFoldersByName(seatStr);
    if (folders.hasNext()) {
      subFolder = folders.next();
    } else {
      subFolder = rootFolder.createFolder(seatStr);
    }

    const timeStampForFile = new Date().getTime();

    // 2. 處理照片上傳
    let photoUrl = "";
    if (data.photoBase64) {
      // 移除 Base64 的 header (例如 data:image/jpeg;base64,)
      const base64Data = data.photoBase64.split(',')[1];
      const photoBlob = Utilities.newBlob(Utilities.base64Decode(base64Data), data.photoMimeType, data.seat + "_照片_" + timeStampForFile);
      const photoFile = subFolder.createFile(photoBlob);
      photoUrl = photoFile.getUrl();
    }

    // 3. 處理簽名圖檔上傳
    let sigUrl = "";
    if (data.sigBase64) {
      const base64Data = data.sigBase64.split(',')[1];
      const sigBlob = Utilities.newBlob(Utilities.base64Decode(base64Data), "image/png", data.seat + "_簽名_" + timeStampForFile + ".png");
      const sigFile = subFolder.createFile(sigBlob);
      sigUrl = sigFile.getUrl();
    }

    // 4. 處理 Google Sheet 表單紀錄 (以當日日期命名)
    const todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
    let dataSheet = ss.getSheetByName(todayStr);
    
    // 如果今天的工作表還沒建立，就新建一個並加上標題
    if (!dataSheet) {
      dataSheet = ss.insertSheet(todayStr);
      dataSheet.appendRow(['紀錄時間', '學生座號', '作業完成度', '情緒表現', '溝通留言', '照片連結', '家長簽名連結']);
      dataSheet.getRange('A1:G1').setFontWeight('bold').setBackground('#e2f0d9');
      dataSheet.setFrozenRows(1);
    }

    // 5. 寫入資料
    const recordTime = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
    dataSheet.appendRow([
      recordTime, 
      data.seat, 
      data.homework, 
      data.emotion, 
      data.comment, 
      photoUrl, 
      sigUrl
    ]);

    return { success: true };
  } catch (e) {
    return { success: false, error: e.toString() };
  }
}

/**
 * 讓前端取得初始資料 (班級、姓名、題庫)
 */
function getInitialData() {
  const { qSheet, unitName } = autoInitSystem();
  const data = qSheet.getDataRange().getValues();
  
  let classes = new Set();
  let students = {};
  let questions = [];

  // 解析資料 (略過標題列 i=0)
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const className = row[0];
    const studentName = row[1];
    
    // 處理班級與學生名單
    if (className && studentName) {
      classes.add(className);
      if (!students[className]) students[className] = [];
      students[className].push(studentName);
    }
    
    // 處理題庫
    const qId = row[5];
    const qType = row[6];
    const qContent = row[7];
    const qOptions = row[8];
    const qAnswer = row[9];
    
    if (qId && qType && qContent) {
      questions.push({
        id: qId,
        type: qType,
        content: qContent,
        options: qOptions ? qOptions.split(',').map(s => s.trim()) : [],
        answer: String(qAnswer).trim()
      });
    }
  }

  return {
    unitName: unitName,
    classes: Array.from(classes),
    students: students,
    questions: questions
  };
}

 * 將手寫圖片上傳至 Google Drive

function uploadImageToDrive(base64Str, filename) {
  if (!base64Str) return "無圖檔";
  
  const props = PropertiesService.getScriptProperties();
  const folderId = props.getProperty('FOLDER_ID');
  
  let folder;
  try {
    folder = folderId ? DriveApp.getFolderById(folderId) : DriveApp.getRootFolder();
  } catch (e) {
    folder = DriveApp.getRootFolder(); // 發生錯誤時退回根目錄
  }

  // 解析 base64
  const split = base64Str.split(',');
  if (split.length < 2) return "圖檔格式錯誤";
  
  const type = split[0].match(/:(.*?);/)[1];
  const data = split[1];
  const blob = Utilities.newBlob(Utilities.base64Decode(data), type, filename);
  
  const file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  
  return file.getUrl();
}

function submitExam(payload) {

  let rowData = [new Date(), className, studentName, score];
  
  // 加入每一題的作答 (限制前10題，避免陣列過長，可依需求調整)
  answers.forEach(ans => rowData.push(ans || "未作答"));
  
  // 補足空欄位，確保圖片與回饋欄位位置一致 (假設最多 3 題)
  while(rowData.length < 7) { rowData.push(""); }
  
  // 加入圖片網址與預留的回饋欄位
  rowData.push(imgUrl, "", "", "");
  
  examSheet.appendRow(rowData);
  return { success: true, message: `繳交成功！您的選擇題得分為：${score} 分。`, score: score };
}

function getFeedback(payload) {
  const { unitName, className, studentName } = payload;
  const { ss } = autoInitSystem();
  const sheet = ss.getSheetByName(unitName);
  if (!sheet) throw new Error("尚未找到考試紀錄表");

  const data = sheet.getDataRange().getValues();
  const headers = data[0];

  const classIdx = headers.indexOf('班級');
  const nameIdx = headers.indexOf('姓名');
  const textIdx = headers.indexOf('教師回饋_文字');
  const linkIdx = headers.indexOf('教師回饋_連結');
  const fileIdx = headers.indexOf('教師回饋_檔案');

  // 從最後一筆資料往上找 (取最新的作答紀錄)
  for (let i = data.length - 1; i >= 1; i--) {
    if (data[i][classIdx] === className && data[i][nameIdx] === studentName) {
      return {
        success: true,
        text: textIdx !== -1 ? data[i][textIdx] : '',
        link: linkIdx !== -1 ? data[i][linkIdx] : '',
        file: fileIdx !== -1 ? data[i][fileIdx] : ''
      };
    }
  }
  return { success: false, error: '尚未找到您的作答紀錄，或教師尚未批改。' };
}

    function setupCanvasEvents() {
        ctx.lineWidth = 4;
        ctx.lineCap = 'round';
        ctx.strokeStyle = '#1e3a8a'; // 深藍色筆跡

        const getPos = (e) => {
            const rect = canvas.getBoundingClientRect();
            const clientX = e.touches ? e.touches[0].clientX : e.clientX;
            const clientY = e.touches ? e.touches[0].clientY : e.clientY;
            // 計算比例以適應 RWD 縮放
            const scaleX = canvas.width / rect.width;
            const scaleY = canvas.height / rect.height;
            return { x: (clientX - rect.left) * scaleX, y: (clientY - rect.top) * scaleY };
        };

        const startDraw = (e) => {
            if (e.cancelable) e.preventDefault(); // 避免手機滑動
            isDrawing = true;
            const pos = getPos(e);
            ctx.beginPath();
            ctx.moveTo(pos.x, pos.y);
        };

        const draw = (e) => {
            if (!isDrawing) return;
            if (e.cancelable) e.preventDefault();
            const pos = getPos(e);
            ctx.lineTo(pos.x, pos.y);
            ctx.stroke();
        };

        const stopDraw = () => { isDrawing = false; ctx.closePath(); };

        canvas.addEventListener('mousedown', startDraw);
        canvas.addEventListener('mousemove', draw);
        canvas.addEventListener('mouseup', stopDraw);
        canvas.addEventListener('mouseout', stopDraw);
        
        canvas.addEventListener('touchstart', startDraw, { passive: false });
        canvas.addEventListener('touchmove', draw, { passive: false });
        canvas.addEventListener('touchend', stopDraw);
    }

    function clearCanvas() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
    }

    function isCanvasBlank() {
        const blank = document.createElement('canvas');
        blank.width = canvas.width;
        blank.height = canvas.height;
        return canvas.toDataURL() === blank.toDataURL();
    }

    function submitExam() {

        const studentName = document.getElementById('nameSelect').value;

        let score = 0;
        let answers = [];
        
        // 批改選擇題並收集答案
        for (let q of globalData.questions) {
            if (q.type === '選擇') {
                const selected = document.querySelector(`input[name="q_${q.id}"]:checked`);
                const val = selected ? selected.value : '';
                answers.push(val);
                if (val === q.answer) {
                    score += (100 / globalData.questions.filter(qt => qt.type === '選擇').length); 
                }
            } else {
                answers.push("見手寫附件");
            }
        }

        // 檢查手寫題

            if (isCanvasBlank() && !confirm("您的手寫區是空白的，確定要直接繳交嗎？")) return;

        }

        // 建立傳送資料

            unitName: globalData.unitName,
            className, studentName, 
            score: Math.round(score), 
            answers, 
            canvasBase64

在校務或行政日常中，我們不可能隨時盯著 Google 試算表看有沒有新資料進來。透過將 GAS 與台灣人最常用的 LINE 結合，只要有人提交前端表單（例如：學生請假、公物報修、午餐登記），系統就會立刻推播訊息到您的手機或專屬的行政群組！
這不僅能做到「零時差」的管理，還能大幅降低資訊漏接的風險，讓您的系統服務升級到完美。

⚙️ 連動 GAS 和 LINE 的重要節點

  • 1
    取得專屬通行證 (Token) 前往 LINE 官方後台申請一個 Token，這就像是一把專屬鑰匙，賦予您的程式「發送訊息到特定聊天室」的合法權限。
  • 2
    大腦誕生 (部署 GAS) 將這把 Token 鑰匙放進 GAS 程式碼中並「發布部署」，您會獲得一串專屬的網址。這串網址就像是您在雲端建立的「智慧收發室」。
  • 3
    建立通訊橋樑 (填寫 Webhook) 進入 LINE 的開發者後台，找到 Webhook URL 欄位，把剛剛取得的 GAS 網址貼上去。
    💡 意義：這就等於告訴 LINE：「嘿！以後只要有新資料進來，請立刻送到這個網址，交給我的 GAS 大腦處理並發送通知！」
  • 4
    達成零時差推播 橋樑接通後！只要家長或學生在前台按下「送出表單」，GAS 就會自動透過這條專屬通道，將整理好的通知瞬間推播到您的 LINE 手機裡。
